טָמֵא בַגּוֹיִם וַיְבָרֵךְ עַל נְטִילַת יָדַיִם
ספר היראה
יה
הוֹצִין בִנְטִילַת יָדַיִם וְיָנְבִיהֶם אַחַר וִיבָרֵךְ עַל כָּל דָבָר לְפָנָיו ולאחריו
נְטִילָה שֶׁלֹא יַחְזְרוּ מַיִם הָרִאשׁוֹנִים בְּרָכָה הָאֲמוּרָה בָּה עַל פִי רַבּוֹתֵנוּ
וְיִטַמְאוּ הַיָּדַיִם וְיִנְגְבָם הֵיטֵב כִּי כָל עה וְאִם אֵינוֹ בָקִי יֵלֶךְ אֵצֶל בְּקִי
הָאוֹכֵל בְּלֹא נִיגוב ידים כאלו כְּדֵי לִלְמוֹד וְאֶל יָשִׂים עַצְמוֹ
אוֹכֵל לֶחֶם טָמֵא שְׁנָאמַר לַחֲטָם כַבְהֵמָה וְאִם לֹא מָצָא בָקִי יָנַיִךְ
הנאתוֹ וְאַל יהנה מן הדבר עד
לאכול וִיבָרֵךְ עַל הַלֶּחֶם אֲשֶׁר יָדַע לְבָרֵךְ עָלָיו אֶת בּוֹרְאוֹ
הַמוֹצִיא לֶחֶם מִן הָאָרֶץ: וְיִתֵּן רֵיוָה כּדָת וְכַמִּשְׁפָּט וּכְשִׁיבָרֵךְ יְכַאֵין אֶת
בֵּין תֵּיבָה לְהֵיבָה וְכָל מַה שֶׁיֶהְנֶה לִבּוֹ לְבָרֵךְ אֶת הַבּוֹרֵא וְלֹא כְּמוֹ
בּוֹ מֵהָעוֹלָם הַזֶּה בֵּין בְּפִירוֹת בֵּין דָבָר הָרָגִיל בְפִי האדם לצאת שלא
בְּרֶיוָה בֵּין בַּאֲכִילָה בֵּין בְּכָל דָבָר בְכַוָּנָה: וְכֵן לְבִרְכַת הַמַּצות וּלְכָל
שיהיה אסוּר לִיהָנוֹת בְּלֹא בְרָכָה מַעֲשֶׂה הָאָדָם יַעֲשֶׂה דְפָרִים לְשֵׁם
